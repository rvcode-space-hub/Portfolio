



"use client";
import React, { useRef, useState } from "react";
import { motion, AnimatePresence } from "framer-motion";
import Image from "next/image";
import certification from "../utils/certification.json";
import { ChevronRight, ChevronLeft, X , Calendar} from "lucide-react";

export default function CertificationsSection() {
  const [activeCard, setActiveCard] = useState(null);
  const [preview, setPreview] = useState(null);
  const scrollRef = useRef(null);

  const scroll = (direction) => {
    scrollRef.current?.scrollBy({
      left: direction === "left" ? -400 : 400,
      behavior: "smooth",
    });
  };
  

  return (
    <div className="w-full py-20 px-6 md:px-10 relative">

      {/* Heading */}
      <div className="text-center mb-16">
        <h2 className="text-3xl md:text-5xl font-bold text-white">
          Company<span className="text-blue-500"> Certifications</span>
        </h2>
      </div>

      {/* Arrows */}
      <div
        onClick={() => scroll("left")}
        className="absolute left-2 top-1/2 -translate-y-1/2 z-10 cursor-pointer bg-white/90 hover:bg-white p-3 rounded-full shadow-md"
      >
        <ChevronLeft size={24} />
      </div>

      <div
        onClick={() => scroll("right")}
        className="absolute right-2 top-1/2 -translate-y-1/2 z-10 cursor-pointer bg-white/90 hover:bg-white p-3 rounded-full shadow-md"
      >
        <ChevronRight size={24} />
      </div>

      {/* Cards */}
      <div
        ref={scrollRef}
        className="flex gap-6 overflow-x-auto scroll-smooth no-scrollbar"
      >
        {certification.map((card) => {
          const isActive = activeCard === card.id;


          return (
            <motion.div
              key={card.id}
              whileHover={{ y: -6 }}
              onClick={() => setPreview(card)} // 👈 open modal
              className={`min-w-[340px] md:min-w-[400px] group cursor-pointer rounded-2xl p-4
              bg-white/5 backdrop-blur-md border border-white/10
              transition-all duration-300
              ${
                isActive
                  ? "scale-105 shadow-[0_0_40px_rgba(59,130,246,0.6)]"
                  : "hover:border-blue-400 hover:shadow-[0_0_25px_rgba(59,130,246,0.4)]"
              }`}
            >

           
              {/* Image */}
              <div className="relative rounded-xl overflow-hidden mb-4">
                <div className="relative w-full h-52">
                  <Image
                    src={card.imgUrl}
                    alt={card.title}
                    fill
                    className="object-cover group-hover:scale-110 transition duration-300"
                  />
                </div>
                <div className="absolute inset-0 bg-gradient-to-t from-black/50 to-transparent" />
              </div>

 <span className="text-blue-400 text-sm flex items-center gap-2 mb-2">
                      <Calendar size={16} />
                      {card.date}
                    </span>
              {/* Content */}
              <h3 className="text-lg font-semibold text-blue-400 mb-1 line-clamp-1">
                {card.title}
              </h3>

              <p className="text-gray-400 text-sm mb-3 line-clamp-2">
                {card.description}
              </p>

              <div className="flex flex-wrap gap-2">
                {card.tags?.map((tag, i) => (
                  <span
                    key={i}
                    className="px-2 py-1 text-xs rounded-full bg-blue-500/10 border border-blue-400/20 text-blue-300"
                  >
                    {tag}
                  </span>
                ))}
              </div>
            </motion.div>
          );
        })}
      </div>

      {/* 🔥 MODAL */}
     <AnimatePresence>
  {preview && (
    <motion.div
      className="fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center"
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0 }}
    >
      {/* Close Button */}
      <button
        onClick={() => setPreview(null)}
        className="absolute top-6 right-6 text-white bg-white/10 hover:bg-white/20 p-2 rounded-full"
      >
        <X size={24} />
      </button>

      {/* Modal Card */}
      <motion.div
        initial={{ scale: 0.85, opacity: 0 }}
        animate={{ scale: 1, opacity: 1 }}
        exit={{ scale: 0.85, opacity: 0 }}
        className="w-[95%] md:w-[80%] lg:w-[70%] max-h-[90vh] bg-[#0B0F19] rounded-2xl overflow-hidden shadow-2xl"
      >
        
        {/* IMAGE FIX */}
        <div className="relative w-full h-[60vh] bg-black flex items-center justify-center">
          <Image
            src={preview.imgUrl}
            alt={preview.title}
            fill
            className="object-contain p-4"
          />
        </div>

        {/* CONTENT */}
        <div className="p-6 border-t border-white/10">
          <h3 className="text-xl md:text-2xl text-blue-400 font-semibold">
            {preview.title}
          </h3>

          <p className="text-gray-400 mt-2 text-sm md:text-base">
            {preview.description}
          </p>
        </div>

      </motion.div>
    </motion.div>
  )}
</AnimatePresence>

    </div>
  );
}





  {/* RIGHT (IMAGE) */}
        <motion.div
          key={activeCard.imgUrl}
          initial={{ opacity: 0, x: 40 }}
          animate={{ opacity: 1, x: 0 }}
        >
          <div className="relative w-full h-[280px] md:h-[420px] lg:h-[480px] rounded-2xl overflow-hidden shadow-2xl border border-white/10">
            <Image
              src={activeCard.imgUrl}
              alt={activeCard.title}
              fill
              className="object-contain p-4 hover:scale-105 transition"
            />
          </div>
        </motion.div>

        {/* Arrows */}
        <button
          onClick={prev}
          className="absolute left-0 top-1/2 -translate-y-1/2 bg-white/90 hover:bg-white p-3 rounded-full"
        >
          <ChevronLeft />
        </button>
